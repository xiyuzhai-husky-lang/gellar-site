contract gather (out : int*) (src : int*) (idx : int*) (n : int) where
  requires n ≤ gridDim * blockDim
  requires gridDim * blockDim ≤ 2^31
  ensures  out[i] = src[idx[i]]   when 0 ≤ i < n
  ensures  out[i] = old(out[i])  otherwise
  unchanged src
  unchanged idx

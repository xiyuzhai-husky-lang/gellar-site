contract vadd (c : int*) (a : int*) (b : int*) (n : int) where
  requires n ≤ gridDim * blockDim
  requires gridDim * blockDim ≤ 2^31
  requires int32 (a[i] + b[i])  when 0 ≤ i < n
  ensures  c[i] = a[i] + b[i]   when 0 ≤ i < n
  ensures  c[i] = old(c[i])     otherwise
  unchanged a
  unchanged b

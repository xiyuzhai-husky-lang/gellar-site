contract warp_sum (out : int*) (src : int*) (n : int) where
  requires blockDim = 32
  requires gridDim * 32 ≤ 2^31
  requires -2^26 ≤ src[i] ∧ src[i] < 2^26  when 0 ≤ i < n
  ensures  out[b] = sum j < 32, (if b * 32 + j < n then src[b * 32 + j] else 0)  when 0 ≤ b < gridDim
  ensures  ∀ (i : Int), (i < 0 ∨ gridDim ≤ i) → out[i] = old(out[i])
  unchanged src
